#!/usr/bin/env python3
"""
Análisis de Accesibilidad Web - FisiChecker
Fórmula de Hilera et al. (Excluyendo NA del denominador)
"""

# ==========================================
# 1. SETUP BASE & INSTALACIÓN DE LIBRERÍAS
# ==========================================
from google.colab import drive
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from pathlib import Path
from urllib.parse import urlparse
from collections import Counter

# Montar Google Drive si tus archivos están en Drive
drive.mount('/content/drive')

# Instalar dependencias necesarias
!pip install -q pandas openpyxl matplotlib seaborn xlsxwriter

# Configuración de estilo visual para gráficos
plt.style.use('seaborn-v0_8-darkgrid')
sns.set_palette("husl")

# ==========================================
# 2. RUTA DEL ARCHIVO EXPORTADO
# ==========================================
# Ajusta esta ruta con el nombre de tu archivo exportado desde FisiChecker
EXCEL_PATH = Path("/content/drive/MyDrive/FisiChecker/auditorias_detalladas.xlsx")

# ==========================================
# 3. CARGA Y ANÁLISIS DE DATOS
# ==========================================
print("Cargando hojas de Excel...")
df_resumen = pd.read_excel(EXCEL_PATH, sheet_name="Resumen")
df_criterios = pd.read_excel(EXCEL_PATH, sheet_name="Criterios WCAG")

print("--- Resumen de Auditorías ---")
print(df_resumen.head())

# Cálculo de Estadísticas Globales por Veredicto
verdict_counts = df_criterios['Veredicto'].value_counts()
print("\nDistribución de Veredictos:")
print(verdict_counts)

# Gráfico de Distribución de Veredictos
plt.figure(figsize=(8, 5))
sns.countplot(data=df_criterios, x='Veredicto', order=['pass', 'fail', 'partial', 'na'], palette='viridis')
plt.title('Distribución Global de Veredictos WCAG 2.1')
plt.xlabel('Veredicto')
plt.ylabel('Cantidad de Evaluaciones')
plt.tight_layout()
plt.show()

# Ranking de Criterios con Mayor Falla
failing_criteria = df_criterios[df_criterios['Veredicto'] == 'fail']['Criterio'].value_counts().head(10)
plt.figure(figsize=(10, 6))
failing_criteria.plot(kind='barh', color='#e11d48')
plt.title('Top 10 Criterios WCAG con Mayor Incumplimiento')
plt.xlabel('Frecuencia de Fallos')
plt.ylabel('Código de Criterio')
plt.gca().invert_yaxis()
plt.tight_layout()
plt.show()

print("¡Análisis completado exitosamente!")
